version: '3.8'

services:
  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data
    networks:
      - transcribe_net

  postgres:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: transcribe_db
      POSTGRES_USER: transcribe_user
      POSTGRES_PASSWORD: transcribe_pass
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - transcribe_net

  api:
    build: .
    ports:
      - "8000:8000"
    environment:
      - DATABASE_URL=postgresql://transcribe_user:transcribe_pass@postgres:5432/transcribe_db
      - REDIS_URL=redis://redis:6379
      - S3_ENDPOINT=https://your-account.r2.cloudflarestorage.com
      - S3_REGION=auto
      - S3_BUCKET=transcribe-storage
      - S3_ACCESS_KEY=${S3_ACCESS_KEY}
      - S3_SECRET_KEY=${S3_SECRET_KEY}
      - PUBLIC_BASE_URL=http://localhost:8000
      - COOKIE_SIGNING_KEY=${COOKIE_SIGNING_KEY:-default-dev-key-change-in-prod}
      - WHISPER_MODEL=medium
      - WHISPER_BEAM_SIZE=5
      - ENABLE_DIARIZATION=true
      - MAX_SPEAKERS=10
      - MAX_DURATION_MIN=15
      - DAILY_CAP_MIN=10
      - LOG_LEVEL=INFO
      - PROMETHEUS_ENABLE=true
      - HUMAN_VERIFICATION=off
    volumes:
      - ./app:/app
      - temp_storage:/tmp/transcribe
    depends_on:
      - redis
      - postgres
    networks:
      - transcribe_net
    command: uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload

  worker:
    build: .
    environment:
      - DATABASE_URL=postgresql://transcribe_user:transcribe_pass@postgres:5432/transcribe_db
      - REDIS_URL=redis://redis:6379
      - S3_ENDPOINT=https://your-account.r2.cloudflarestorage.com
      - S3_REGION=auto
      - S3_BUCKET=transcribe-storage
      - S3_ACCESS_KEY=${S3_ACCESS_KEY}
      - S3_SECRET_KEY=${S3_SECRET_KEY}
      - WHISPER_MODEL=medium
      - WHISPER_BEAM_SIZE=5
      - ENABLE_DIARIZATION=true
      - MAX_SPEAKERS=10
      - LOG_LEVEL=INFO
    volumes:
      - ./app:/app
      - temp_storage:/tmp/transcribe
    depends_on:
      - redis
      - postgres
    networks:
      - transcribe_net
    command: celery -A app.tasks.main worker --loglevel=info --concurrency=2 --queues=free

  traefik:
    image: traefik:v3.0
    command:
      - "--api.insecure=true"
      - "--providers.file.directory=/etc/traefik/dynamic"
      - "--providers.file.watch=true"
      - "--entrypoints.web.address=:80"
    ports:
      - "80:80"
      - "8080:8080"  # Traefik dashboard
    volumes:
      - ./infra/local/traefik.toml:/etc/traefik/dynamic/traefik.toml
    networks:
      - transcribe_net

volumes:
  redis_data:
  postgres_data:
  temp_storage:

networks:
  transcribe_net:
    driver: bridge
